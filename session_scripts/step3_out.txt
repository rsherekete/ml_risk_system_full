WITH cte_transactions AS (
    SELECT
        id,
        account_id,
        transaction_datetime,
        transaction_date_london_trading_day,
        usd_amount,
        completed_transaction_type,
        funds_account_type
    FROM `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_transactions`
    WHERE status = 'Paid'
),

cte_transactions_agg AS (
    SELECT
        account_id,
        SUM(
            IF(
                completed_transaction_type = 'Deposit',
                usd_amount,
                0
            )
        ) AS total_deposit_usd,
        MAX(
            IF(
                completed_transaction_type IN ('Withdrawal'),
                transaction_datetime,
                NULL
            )
        ) AS last_withdrawal_date,
        MAX_BY(
            CASE
                WHEN completed_transaction_type IN ('Withdrawal', 'IB Withdrawal')
                    THEN usd_amount
                ELSE 0
            END,
            CASE
                WHEN completed_transaction_type IN ('Withdrawal', 'IB Withdrawal')
                    THEN transaction_datetime
            END
        ) AS last_withdrawal_amount_usd,
        SUM(
            IF(
                completed_transaction_type IN ('Withdrawal', 'IB Withdrawal')
                AND funds_account_type = 'MT5',
                usd_amount,
                0
            )
        ) AS total_withdrawal_usd,
        SUM(
            IF(
                completed_transaction_type IN ('Withdrawal', 'IB Withdrawal')
                AND funds_account_type = 'Ewallet',
                usd_amount,
                0
            )
        ) AS total_ib_withdrawal_usd,
        SUM(
            IF(
                funds_account_type = 'Ewallet',
                usd_amount,
                0
            )
        ) AS wallet_balance_usd
    FROM cte_transactions
    GROUP BY 1
),

tr_bl AS (
    SELECT
        account_id,
        SUM(trading_balance_usd) AS trading_balance_usd,
        MAX(last_trade_datetime) AS last_trade_datetime,
        SUM(COALESCE(total_volume, 0)) AS total_volume,

    ON fu.referral_link_id = l.id

        SAFE_CAST(fu.last_deposit_id AS INT64) = SAFE_CAST(t_last.id AS INT64)

LEFT JOIN tr_bl

LEFT JOIN cte_utms ON fu.id = cte_utms.id
LEFT JOIN `zfx-dwh-prod`.`views`.`marketing_acquisition_traffic_types` AS aq

        admin_fee,
        -- TS markups are calculated at the deal level; therefore, we sum them up for an order, 
        -- taking into account cases where the number of opening and closing orders is not equal. 
        -- In this case, opening markups are distributed among the closing orders.
        -- cnt_clsd_orders_in_position is for opening side only.
        close_ts_markup,
        close_ts_usd_qty,
        open_ts_markup / cnt_clsd_orders_in_position AS open_ts_markup,
        open_ts_usd_qty / cnt_clsd_orders_in_position AS open_ts_usd_qty,
        -- IB markups are calculated at the order level; therefore, we do not need them to sum up.
        -- And cnt_clsd_orders_in_position is for both sides.
        -- The direction should be taken into account.
        IF(is_buy = 1, -1, 1) * open_ib_markup / cnt_clsd_orders_in_position AS open_ib_markup,
        IF(is_buy = 0, -1, 1) * close_ib_markup / cnt_clsd_orders_in_position AS close_ib_markup,
        -- one position can include more than one closing orders (typical behaviour for CLOSE_BY). 
        -- In this case we split open commissions/swaps between them
        open_commission / cnt_clsd_orders_in_position + close_commission AS commission,
        open_storage / cnt_clsd_orders_in_position + close_storage AS storage,
        -- price_position = 0 AND profit = 0 is a CLOSE_BY case. 
        -- Means that the whole opening volume was closed by CLOSE_BY order, and prices/profits are considered in the 
        -- position what was opened instead of the previous one. In this case price_position = 0 AND profit = 0, 
        -- and we take a close_price as an open one to explain zero profit.
        IF(price_position = 0 AND profit = 0, price, price_position) AS open_price,
        -- taken into account CLOSE BY cases for original_client_price
        IF(price_position = 0 AND profit = 0, NULL, open_original_client_price) AS open_original_client_price,
        IF(order_type = 8, NULL, close_original_client_price) AS close_original_client_price

            AS close_time_london_trading_date,

        -- calculation as per this doc https://zealgroup.sg.larksuite.com/wiki/A729wTaJqibg0OkfoGSlJmtqgzs
        CASE
        WHEN REGEXP_CONTAINS(UPPER(ta.trading_conditions), r'(ECN|PREMIER|PRO)') THEN o.volume_closed / 10000
        WHEN UPPER(ta.trading_conditions) LIKE '%STP%' AND NOT REGEXP_CONTAINS(o.symbol, r'(US30t|DAX30)') THEN o.volume_closed / 10000
        WHEN UPPER(ta.trading_conditions) LIKE '%STP%' AND o.symbol LIKE '%US30t%' THEN o.volume_closed / 10000 / 10
        WHEN UPPER(ta.trading_conditions) LIKE '%STP%' AND o.symbol LIKE '%DAX30%' THEN o.volume_closed / 10000 / 25
        WHEN UPPER(ta.trading_conditions) LIKE '%CENT%' AND NOT REGEXP_CONTAINS(o.symbol, r'(US30t|DAX30)') THEN o.volume_closed / 10000 / 100
        WHEN UPPER(ta.trading_conditions) LIKE '%CENT%' AND o.symbol LIKE '%US30t%' THEN o.volume_closed / 10000 / 100 / 10
        WHEN UPPER(ta.trading_conditions) LIKE '%CENT%' AND o.symbol LIKE '%DAX30%' THEN o.volume_closed / 10000 / 100 / 25
        ELSE o.volume_closed / 10000
    END AS volume_std_lots,
        o.volume_closed / 10000 * sh.contract_size AS qty,
        CASE
            -- FOREX
            WHEN
                s.calc_mode IN (0, 5) AND UPPER(g.currency) != 'CNT'
                THEN (o.volume_closed / 10000) * sh.contract_size * ra.mid
            -- CFD
            WHEN
                s.calc_mode IN (2, 3, 4) AND UPPER(g.currency) != 'CNT'
                THEN (o.volume_closed / 10000) * o.close_price * sh.contract_size * ra.mid
            -- FUTURES
            WHEN
                s.calc_mode = 1 AND UPPER(g.currency) != 'CNT'
                THEN (o.volume_closed / 10000) * o.close_price * ra.mid * (sh.tick_value / sh.tick_size)
            WHEN
                s.calc_mode IN (0, 5) AND UPPER(g.currency) = 'CNT'
                THEN (o.volume_closed / 10000) * sh.contract_size * ra.mid * 0.01
            WHEN
                s.calc_mode IN (2, 3, 4) AND UPPER(g.currency) = 'CNT'
                THEN (o.volume_closed / 10000) * o.close_price * sh.contract_size * ra.mid * 0.01
            WHEN
                s.calc_mode = 1 AND UPPER(g.currency) = 'CNT'
                THEN (o.volume_closed / 10000) * o.close_price * ra.mid * (sh.tick_value / sh.tick_size) * 0.01
        END AS close_qty_usd,
        CASE
            -- FOREX
            WHEN
                s.calc_mode IN (0, 5) AND UPPER(g.currency) != 'CNT'
                THEN (o.volume_closed / 10000) * sh.contract_size * rao.mid
            -- CFD
            WHEN
                s.calc_mode IN (2, 3, 4) AND UPPER(g.currency) != 'CNT'
                THEN (o.volume_closed / 10000) * o.open_price * sh.contract_size * rao.mid
            -- FUTURES
            WHEN
                s.calc_mode = 1 AND UPPER(g.currency) != 'CNT'
                THEN (o.volume_closed / 10000) * o.open_price * rao.mid * (sh.tick_value / sh.tick_size)
            WHEN
                s.calc_mode IN (0, 5) AND UPPER(g.currency) = 'CNT'
                THEN (o.volume_closed / 10000) * sh.contract_size * rao.mid * 0.01
            WHEN
                s.calc_mode IN (2, 3, 4) AND UPPER(g.currency) = 'CNT'
                THEN (o.volume_closed / 10000) * o.open_price * sh.contract_size * rao.mid * 0.01
            WHEN
                s.calc_mode = 1 AND UPPER(g.currency) = 'CNT'
                THEN (o.volume_closed / 10000) * o.open_price * rao.mid * (sh.tick_value / sh.tick_size) * 0.01
        END AS open_qty_usd,
        CASE
            -- FOREX
            WHEN
                s.calc_mode IN (0, 5) AND UPPER(g.currency) != 'CNT'
                THEN
                    ((o.volume_closed / 10000) * sh.contract_size * rao.mid)
                    + ((o.volume_closed / 10000) * sh.contract_size * ra.mid)
            -- CFD
            WHEN
                s.calc_mode IN (2, 3, 4) AND UPPER(g.currency) != 'CNT'
                THEN
                    ((o.volume_closed / 10000) * o.close_price * sh.contract_size * rao.mid)
                    + ((o.volume_closed / 10000) * o.open_price * sh.contract_size * ra.mid)
            -- FUTURES
            WHEN
                s.calc_mode = 1 AND UPPER(g.currency) != 'CNT'
                THEN
                    ((o.volume_closed / 10000) * o.close_price * rao.mid * (sh.tick_value / sh.tick_size))
                    + ((o.volume_closed / 10000) * o.open_price * ra.mid * (sh.tick_value / sh.tick_size))
            WHEN
                s.calc_mode IN (0, 5) AND UPPER(g.currency) = 'CNT'
                THEN
                    ((o.volume_closed / 10000) * sh.contract_size * rao.mid * 0.01)
                    + ((o.volume_closed / 10000) * sh.contract_size * ra.mid * 0.01)
            WHEN
                s.calc_mode IN (2, 3, 4) AND UPPER(g.currency) = 'CNT'
                THEN
                    ((o.volume_closed / 10000) * o.close_price * sh.contract_size * rao.mid * 0.01)
                    + ((o.volume_closed / 10000) * o.open_price * sh.contract_size * ra.mid * 0.01)
            WHEN
                s.calc_mode = 1 AND UPPER(g.currency) = 'CNT'
                THEN
                    ((o.volume_closed / 10000) * o.close_price * rao.mid * (sh.tick_value / sh.tick_size) * 0.01)
                    + ((o.volume_closed / 10000) * o.open_price * ra.mid * (sh.tick_value / sh.tick_size) * 0.01)
        END AS qty_usd, -- BI-2461, closing part changed from volume to volume_closed
        ROUND(o.open_price, CAST(sh.digits AS INT64)) AS open_price,
        ROUND(o.open_original_client_price, CAST(sh.digits AS INT64)) AS open_original_client_price,
        ROUND(o.close_price, CAST(sh.digits AS INT64)) AS close_price,
        ROUND(o.close_original_client_price, CAST(sh.digits AS INT64)) AS close_original_client_price,
        o.profit AS trade_profit,
        ROUND(o.profit * rg.mid, CAST(s.currency_profit_digits AS INT64)) AS trade_profit_usd,
        o.commission * rg.mid AS commission_usd,
        o.storage * rg.mid AS swap_usd,
        o.admin_fee * rg.mid AS admin_fee_usd,
        o.price_sl AS sl,
        o.price_tp AS tp,
        o.comment,
        -- confirmed with Oliver to take current groups in cases when historical data not available, ticket: TDV-1786
        COALESCE(tago.trading_account_group_name, ta.trading_group) AS open_group,
        COALESCE(tagc.trading_account_group_name, ta.trading_group) AS close_group,
        CASE
            WHEN COALESCE(LOWER(tago.trading_account_group_name), LOWER(ta.trading_group), '') LIKE r'%\\o\\%' THEN 'A'
            WHEN COALESCE(LOWER(tago.trading_account_group_name), LOWER(ta.trading_group), '') LIKE r'%\\r\\%' THEN 'B'
        END AS `open_book`,
        CASE
            WHEN COALESCE(LOWER(tagc.trading_account_group_name), LOWER(ta.trading_group), '') LIKE r'%\\o\\%' THEN 'A'
            WHEN COALESCE(LOWER(tagc.trading_account_group_name), LOWER(ta.trading_group), '') LIKE r'%\\r\\%' THEN 'B'
        END AS `close_book`,
        -- FBR markup + IB markup. Bridge doesn't exist in MT5, hence markup = spread_diff.
        COALESCE(gs_open.spread_diff, 0) AS open_markup,
        -- COALESCE(gs_close.spread_diff, 0) AS close_markup, -- moved to next cte
        COALESCE(gs_open.spread_diff, 0) AS open_spread_diff,
        -- COALESCE(gs_close.spread_diff, 0) AS close_spread_diff, -- moved to next cte

        COALESCE(mdr.rebate_spread_usd, 0) AS rebate_payout_markup,
        0 AS rebate_payout_commission,
        0 AS rebate_payout_subsidy,
        COALESCE(mdr.rebate_spread_usd, 0) AS rebate_payout,
        -- open_source TBD
        NULL AS open_source,
        -- close_source TBD
        NULL AS close_source,
        o.open_market_bid AS open_bid,
        o.open_market_ask AS open_ask,
        o.open_price AS open_confirmed_price,
        -- raw_execution_price = price_gateway. price_gateway = IB markup + deal_price
        o.open_price + o.open_ib_markup AS open_raw_execution_price,
        o.close_market_bid AS close_bid,
        o.close_market_ask AS close_ask,
        o.close_price AS close_confirmed_price,
        -- raw_execution_price = price_gateway. price_gateway = IB markup + deal_price
        o.close_price + o.close_ib_markup AS close_raw_execution_price,
        -- FBR markups. Bridge doesn't exist in mt5 hence markups = 0
        0 AS open_fbr_markup_usd,
        0 AS close_fbr_markup_usd,
        -- IB markup. Price_gateway - Deal Price.
        IF(UPPER(g.currency) = 'CNT', open_ib_markup * 0.01, open_ib_markup) AS open_ib_markup_usd,
        IF(UPPER(g.currency) = 'CNT', close_ib_markup * 0.01, close_ib_markup) AS close_ib_markup_usd,
        -- TS markups. Based on DoneTrades, the same logic as for MT4.
        IF(UPPER(g.currency) = 'CNT', open_ts_markup * 0.01, open_ts_markup) AS open_ts_markup_usd,
        IF(UPPER(g.currency) = 'CNT', close_ts_markup * 0.01, close_ts_markup) AS close_ts_markup_usd,
        --
        s.calc_mode AS profit_mode,
        NULL AS trade_margin,
        -- Execution time in ms
        open_time_ms,
        close_time_ms,
        -- Setup time in ms
        open_time_setup_ms,
        close_time_setup_ms,

        -- FXBO accounts
        CAST(a.id AS STRING) AS `account_id`,
        a.created_at AS `account_registration_datetime`,
        DATETIME(a.created_at, 'Europe/London') AS `account_registration_datetime_london`,
        DATE(a.created_at, 'Europe/London') AS `account_registration_datetime_london_trading_day`,

        a_hist.assigned_user_id AS `account_assgined_user_id_hist`,
        a_hist.assigned_user_name AS `account_assgined_user_name_hist`, -- additional
        CAST(md.department_id AS STRING) AS `account_owner_office_id_hist`,
        d.name AS `account_owner_office_name_hist`,

        -- current values:
        -- , taf.account_owner_id AS `account_assgined_user_id_hist` 
        -- , taf.account_owner_name AS `account_assgined_user_name_hist` 
        -- , taf.account_owner_office_id AS `account_owner_office_id_hist`
        -- , taf.account_owner_office_name AS `account_owner_office_name_hist` 

        -- TODO, this is current parent
        taf.parent_id AS `account_parent_account_id_hist`,
        taf.parent_name AS `account_parent_account_name_hist`,
        SAFE_CAST(NULL AS STRING) AS `account_sales_account_id_hist`,
        taf.account_type AS `account_type_hist`,
        -- TODO, this is current
        NOT COALESCE(a.verified_at IS NULL, FALSE) AS `account_passed_kyc_hist`,

        SAFE_CAST(JSON_EXTRACT_SCALAR(utm_params, '$.utm_campaign') AS STRING) AS `account_utm_campaign`,
        SAFE_CAST(JSON_EXTRACT_SCALAR(utm_params, '$.utm_content') AS STRING) AS `account_utm_content`,
        SAFE_CAST(JSON_EXTRACT_SCALAR(utm_params, '$.utm_medium') AS STRING) AS `account_utm_medium`,
        SAFE_CAST(JSON_EXTRACT_SCALAR(utm_params, '$.utm_source') AS STRING) AS `account_utm_source`,
        SAFE_CAST(JSON_EXTRACT_SCALAR(utm_params, '$.utm_term') AS STRING) AS `account_utm_term`,

        -- Country of residence in FXBO
        a.country AS `account_country_code_hist`,

        -- In FXBO single mode which is the equivalent to mode B 
        SAFE_CAST(NULL AS STRING) AS `account_mode_hist`,

        -- There is no such entity as primary trading account, so client id is used
        SAFE_CAST(a.id AS STRING) AS `primary_trading_account_id`,
        SAFE_CAST(a.id AS INT64) AS `primary_trading_account_number`,

        -- Hash of login is used as there is no trading accs id in FXBO
        SAFE_CAST(ma.login AS STRING) AS `trading_account_id`,

        ma.registration AS `trading_account_registration_datetime`,
        DATETIME(ma.registration, 'Europe/London') AS `trading_account_registration_datetime_london`,
        DATE(ma.registration, 'Europe/London') AS `trading_account_registration_datetime_london_trading_day`,

        -- Trading conditions are current state
        ta.trading_conditions AS `trading_account_conditions_id_hist`,

        -- There is no additional assignment for trading account
        SAFE_CAST(NULL AS STRING) AS `trading_account_assigned_user_id_hist`,
        SAFE_CAST(NULL AS STRING) AS `trading_account_parent_trading_account_id_hist`,
        SAFE_CAST(NULL AS STRING) AS `trading_account_assigned_user_name_hist`,
        -- BI-3865, column path added for the next step:
        s.path AS s_path
    FROM part_3 AS o
    INNER JOIN `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_trading_accounts_flagged` AS taf
        ON
            o.login = taf.login
            AND taf.is_test = 0
            AND taf.is_demo = 0
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_mt5_dubai_live01_accounts` AS ma ON o.login = ma.login
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_mt5_dubai_live01_groups` AS g ON ma.group = g.group
    LEFT JOIN
        `zfx-dwh-prod`.`rep_general`.`tbl_currency_rates` AS rg
        ON g.currency = rg.currency AND rg.rate_date = DATE(o.close_time)
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_mt5_dubai_live01_symbols` AS s ON o.symbol = s.symbol
    LEFT JOIN `zfx-dwh-prod`.`data_marts_multi_region`.`zeal_mt5_consymbol_with_validity_ranges` AS sh
        ON
            o.symbol = sh.symbol
            AND o.close_time >= sh._valid_from_dttm
            AND o.close_time < sh._valid_to_dttm
    LEFT JOIN `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_trading_conditions` AS ta ON g.group = ta.trading_group
    LEFT JOIN `zfx-dwh-prod`.`rep_general`.`tbl_currency_rates` AS ra
        ON
            s.currency_base = ra.currency
            AND ra.rate_date = DATE(o.close_time)
    LEFT JOIN `zfx-dwh-prod`.`rep_general`.`tbl_currency_rates` AS rao
        ON
            s.currency_base = rao.currency
            AND rao.rate_date = DATE(o.open_time)
    LEFT JOIN `zfx-dwh-prod`.`data_marts_multi_region`.`trading_account_group_changes_optimized` AS tago
        ON
            tago.server_name = 'mt5_dubai_live01'
            AND o.login = tago.trading_account_number
            AND o.open_time >= tago.valid_from_timestamp
            AND o.open_time < tago.valid_to_timestamp
    LEFT JOIN `zfx-dwh-prod`.`data_marts_multi_region`.`trading_account_group_changes_optimized` AS tagc
        ON
            tagc.server_name = 'mt5_dubai_live01'
            AND o.login = tagc.trading_account_number
            AND o.close_time >= tagc.valid_from_timestamp
            AND o.close_time < tagc.valid_to_timestamp
    -- for spread_diff
    LEFT JOIN group_symbols AS gs_open
        ON
            tago.trading_account_group_name = gs_open.`group`
            AND STARTS_WITH(s.path, gs_open.path_prefix)
    -- FXBO
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_fxbackoffice_users` AS a
        ON COALESCE(CAST(ma.idx AS STRING), '') = COALESCE(CAST(a.id AS STRING), '')
    LEFT JOIN mt5_dubai_rebates AS mdr ON CAST(o.`order` AS INT64) = CAST(mdr.`order` AS INT64)
    LEFT JOIN `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_accounts_assigned_user_rel_hist` AS a_hist
        ON
            taf.account_id = a_hist.account_id
            AND o.close_time >= a_hist.valid_from_timestamp
            AND o.close_time < a_hist.valid_to_timestamp
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_fxbackoffice_managers_to_departments` AS md
        ON SAFE_CAST(a_hist.assigned_user_id AS STRING) = SAFE_CAST(md.manager_id AS STRING)
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_fxbackoffice_managers_departments` AS d
        ON md.department_id = d.id
    LEFT JOIN closing_deal AS cd ON o.position_id = cd.position_id
    LEFT JOIN opening_deal AS od ON o.position_id = od.position_id
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY o.position_id, o.`order`, s.path, tago.trading_account_group_name
        -- we take the most detailed path. 
        -- For example, if there are two options: 
        -- Forex\FX Major\* and Forex\* 
        -- We will take Forex\FX Major\*
        ORDER BY gs_open.path_len DESC
    ) = 1
),

        WHEN UPPER(ta.trading_conditions) LIKE '%STP%' AND NOT REGEXP_CONTAINS(p.symbol, r'(US30t|DAX30)') THEN p.volume / 10000
        WHEN UPPER(ta.trading_conditions) LIKE '%STP%' AND p.symbol LIKE '%US30t%' THEN p.volume / 10000 / 10
        WHEN UPPER(ta.trading_conditions) LIKE '%STP%' AND p.symbol LIKE '%DAX30%' THEN p.volume / 10000 / 25
        WHEN UPPER(ta.trading_conditions) LIKE '%CENT%' AND NOT REGEXP_CONTAINS(p.symbol, r'(US30t|DAX30)') THEN p.volume / 10000 / 100

WITH blnc AS (
    SELECT
        taf.login,
        taf.trading_account_group,
        acc.leverage,
        acc.comment,
        (d.mid * acc.balance) AS balance,
        (d.mid * acc.credit) AS credit
    FROM `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_trading_accounts_flagged` AS taf
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_mt5_dubai_live01_accounts` AS acc
        ON taf.login = acc.login
    LEFT JOIN `zfx-dwh-prod`.`operational_data_store`.`dubai_traze_mt5_dubai_live01_groups` AS g
        ON taf.trading_account_group = g.group
    LEFT JOIN `zfx-dwh-prod`.`rep_general`.`tbl_currency_rates`
        AS d ON g.currency = d.currency
    AND d.rate_date = CURRENT_DATE()
),

trade_date AS (
    SELECT
        ct.login,
        SUM(volume_std_lots) AS total_volume,
        MAX(open_time) AS last_open_time,
        MAX(close_time) AS last_trade_datetime,
        SUM(
            CASE
                WHEN s.symbol_type = 'FX Spot' THEN ct.volume_std_lots
                ELSE 0
            END
        ) AS total_volume_fx,
        SUM(
            CASE
                WHEN s.symbol_type = 'Metal Spot' THEN ct.volume_std_lots
                ELSE 0
            END
        ) AS total_volume_metal,
        SUM(
            CASE
                WHEN (
                    s.symbol_type IS NULL
                    OR s.symbol_type NOT IN ('FX Spot', 'Metal Spot')
                ) THEN ct.volume_std_lots
                ELSE 0
            END
        ) AS total_volume_cfd,
        SUM(qty_usd) AS qty_usd,
        SUM(trade_profit_usd) AS total_trade_profit_usd,
        SUM(swap_usd) AS total_swaps_usd,
        SUM(commission_usd) AS total_commission_usd,
        SUM(rebate_payout) AS total_rebates_generated_usd,
        SUM(rebate_payout_markup) AS total_rebate_spread,
        SUM(rebate_payout_commission) AS total_rebate_commission,
        SUM(rebate_payout_subsidy) AS total_rebate_subsidy
    FROM
        `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_closed_trades` AS ct
    LEFT JOIN `zfx-dwh-prod`.`rep_general`.`symbols` AS s ON ct.symbol_name = s.symbol
    GROUP BY
        ct.login
),

open AS (
    SELECT
        login,
        MAX(open_time) AS open_time,
        SUM(open_qty_usd) AS open_qty_usd
    FROM
        `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_open_trades`
    GROUP BY
        login
),

float_data AS (
    SELECT
        login,
        last_update,
        SUM(volume_std_lots) AS floating_volume,
        SUM(open_qty_usd) AS floating_volume_usd,
        SUM(floating_trade_profit_usd) AS floating_profit_usd,
        SUM(swap_usd) AS floating_swap_usd,
        SUM(commission_usd) AS floating_commission_usd
    FROM
        `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_open_trades`
    GROUP BY
        login,
        last_update
),

cte_withdrawals AS (
    SELECT
        t.funds_trading_account_number,
        MAX(
            IF(
                completed_transaction_type IN ('Withdrawal'),
                transaction_datetime,
                NULL
            )
        ) AS last_withdrawal_date,
        SUM(
            IF(
                completed_transaction_type = 'Withdrawal',
                t.usd_amount,
                0
            )
        ) AS total_withdrawal_usd
    FROM `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_transactions` AS t
    WHERE completed_transaction_type IN ('Withdrawal')
    GROUP BY
        t.funds_trading_account_number
),

first_depo AS (
    SELECT DISTINCT
        t.funds_trading_account_number,
        MIN(t.transaction_datetime) OVER (PARTITION BY t.funds_trading_account_number) AS first_deposit_date,
        FIRST_VALUE(t.usd_amount) OVER (
            PARTITION BY t.funds_trading_account_number
            ORDER BY
                t.transaction_datetime ASC
        ) AS first_deposit_amount_usd,
        MAX(t.transaction_datetime) OVER (PARTITION BY t.funds_trading_account_number) AS last_deposit_date,
        FIRST_VALUE(t.usd_amount) OVER (
            PARTITION BY t.funds_trading_account_number
            ORDER BY
                t.transaction_datetime DESC
        ) AS last_deposit_amount_usd,
        SUM(t.usd_amount) OVER (PARTITION BY t.funds_trading_account_number) AS total_deposit_usd
    FROM `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_transactions` AS t
    WHERE t.completed_transaction_type = 'Deposit'
),

cte_other_payments AS (
    SELECT
        t.funds_trading_account_number,
        SUM(
            CASE
                WHEN t.completed_transaction_type = 'Zero Balance Adjustment' THEN t.usd_amount
                ELSE 0
            END
        ) AS total_negative_balance_adj_usd
    FROM `zfx-dwh-prod`.`data_marts_multi_region`.`mt5_dubai_live01_transactions` AS t
    GROUP BY t.funds_trading_account_number
)

    CASE

        WHEN fu.client_type = 'Corporate' THEN 'Institutional'

    fu.country AS account_country_code,
    cd.country_name AS country_of_residence,
    cd.continent_region_internal AS region,
    CAST(fu.partner_id AS STRING) AS parent_id,
    CONCAT(TRIM(parent_u.first_name), ' ', TRIM(parent_u.last_name)) AS parent_name,
    CAST(fu.manager_id AS STRING) AS account_owner_id,
    m.full_name AS account_owner_name,
    CAST(md.department_id AS STRING) AS account_owner_office_id,

    acc.registration_utc AS trading_account_registration_datetime,
    DATE(DATETIME(acc.registration_utc + INTERVAL 2 HOUR, 'Europe/London')) AS trading_account_registration_london_trade_date,
    -- logic for test accounts provided by Oliver Hopkins

        WHEN fu.is_employee = 1 THEN 1
        -- WHEN COALESCE(acc.`group`, '') LIKE '%demo%' THEN 1 -- moved to is_demo flag
        WHEN LOWER(COALESCE(acc.`group`, '')) LIKE '%test%' THEN 1
        --- accounts with null id in FXBO are considered as test accounts, according to Oliver
        WHEN (fu.id IS NULL) THEN 1
        --- not %test% to avoid situations when client's name contains test in his first_name
        WHEN LOWER(TRIM(fu.first_name)) = 'test' THEN 1
        --- not %test% to avoid situations when client's name contains test in his last_name
        WHEN LOWER(TRIM(fu.last_name)) = 'test' THEN 1
        ELSE 0
    END AS is_test,
    CASE
        WHEN LOWER(COALESCE(acc.`group`, '')) LIKE '%demo%' THEN 1
        ELSE 0
    END AS is_demo

    ON md.department_id = d.id
